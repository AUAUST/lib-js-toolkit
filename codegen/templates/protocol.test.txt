import { implementsProtocol } from "@auaust/toolkit/protocols";
import { {{symbol}}, type {{interface}} } from "@auaust/toolkit/protocols/{{protocol}}";
import { describe, expect, expectTypeOf, test } from "vitest";

describe("{{protocol}}", () => {
  test("recognizes implementations and narrows their type", () => {
    const value: unknown = {
      [{{symbol}}]: undefined,
    };

    expect(implementsProtocol({{symbol}}, value)).toBe(true);

    if (implementsProtocol({{symbol}}, value)) {
      expectTypeOf(value).toEqualTypeOf<{{interface}}>();
    }
  });
});
